VARIABLE host host.docker.internal
VARIABLE port 8080
# Hard ceiling on total CCSDS packet size in bytes. Enforced before the rules,
# and deliberately NOT editable from the rules file.
VARIABLE fw_max_length 1024

TARGET EAGLEGATE <%= eaglegate_target_name %>
  # Decom tlm and cmd data in the time-series database is kept forever by default.
  # Uncomment to enforce a 30 day retention time. See https://docs.openc3.com/docs/configuration/plugins
  # CMD_DECOM_RETAIN_TIME 30d
  # TLM_DECOM_RETAIN_TIME 30d

INTERFACE <%= eaglegate_target_name %>_INT openc3/interfaces/tcpip_client_interface.py <%= host %> <%= port %> <%= port %> 10.0 None
  MAP_TARGET <%= eaglegate_target_name %>
  # No LENGTH on the INTERFACE line: the firewall protocol does the CCSDS framing
  # itself, so it can check each header before buffering the packet body.
  # Arguments: size ceiling, rules file. Change rules at runtime with the
  # EAGLEGATE/procedures/apply_firewall_rules.py procedure.
  PROTOCOL READ eaglegate_firewall_protocol.py <%= fw_max_length %> <%= eaglegate_target_name %>/rules/firewall_rules.json
  # Key for "authenticity" rules (hex). Create the secret in Admin > Secrets,
  # then uncomment so the rule's "key_env" can read it:
  # SECRET ENV EAGLEGATE_AUTH_KEY EAGLEGATE_AUTH_KEY
